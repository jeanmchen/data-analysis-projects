-- contents of stored procedure:

    USE `employee`;
    DROP procedure IF EXISTS `YEARS_EXPERIENCE`;

    DELIMITER $$
    USE `employee`$$
    CREATE PROCEDURE `YEARS_EXPERIENCE` (yearsWanted int)
    BEGIN
        select EMP_ID, FIRST_NAME, LAST_NAME, ROLE, DEPT, EMP_RATING, EXP
            from EMP_RECORD_TABLE 
            where EXP >= yearsWanted;
    END$$

    DELIMITER ;

-- usage of stored procedure:

call YEARS_EXPERIENCE (3);

-- contents of function:

    DELIMITER $$
    CREATE DEFINER=`root`@`localhost` FUNCTION `Emp_Audit`(eid char(4)) RETURNS varchar(50) CHARSET utf8mb4
        DETERMINISTIC
    BEGIN
        declare ex int;
        declare vRole varchar(100);
        declare flag varchar(10);
        select exp, role into ex, vRole from data_science_team where emp_id = eid;
            
            if ex > 12 and ex <= 16 then
                if vRole = 'Manager' then
                    set flag = 'Yes';
                else
                    set flag = 'Not';
                end if;
                
            elseif ex > 10 and ex <= 12 then
                if vRole = 'Lead Data Scientist' then
                    set flag = 'Yes';
                else
                    set flag = 'Not';
                end if;

            elseif ex > 5 and ex <= 10 then
                if vRole = 'Senior Data Scientist' then
                    set flag = 'Yes';
                else
                    set flag = 'Not';
                end if;  
                
            elseif ex > 2 and ex <= 5 then
                if vRole = 'Associate Data Scientist' then
                    set flag = 'Yes';
                else
                    set flag = 'Not';
                end if;              
                
            elseif ex <= 2 then
                if vRole = 'Junior Data Scientist' then
                    set flag = 'Yes';
                else
                    set flag = 'Not';
                end if;              
        end if;
    RETURN flag;
    END$$

    DELIMITER ;
    ;

-- usage of function:

select *, emp_audit(emp_id) from data_science_team;
